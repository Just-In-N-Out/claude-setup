	Remove []ItemKey

// ApplyChange returns the next version's items and never modifies parent.
// An empty change, removing an item the parent lacks, naming an item twice,
// or adding one that is still present after the removals is invalid.
func ApplyChange(parent []SeedItem, change Change) ([]SeedItem, error) {
	if len(change.Add) == 0 && len(change.Remove) == 0 {
		return nil, invalid("change")
	}
	present := make(map[ItemKey]bool, len(parent))
	for _, item := range parent {
		present[item.Key()] = true
	}
	removing := make(map[ItemKey]bool, len(change.Remove))
	for _, key := range change.Remove {
		key.Name = strings.TrimSpace(key.Name)
		if !key.Kind.Valid() || !present[key] || removing[key] {
			return nil, invalid("item to remove")
		}
		removing[key] = true
	}
	next := make([]SeedItem, 0, len(parent)+len(change.Add))
	kept := make(map[ItemKey]bool, len(parent)+len(change.Add))
	for _, item := range parent {
		if !removing[item.Key()] {
			next = append(next, item)
			kept[item.Key()] = true
		}
	}
	for _, item := range change.Add {
		if kept[item.Key()] {
			return nil, invalid("item to add")
		}
		kept[item.Key()] = true
		next = append(next, item)
	}
	return next, nil
}

// SetPackBuilder binds the pack builder during startup, before serving requests.

// packGuard protects the JSON pack builder API: a state-changing request must
// carry the CSRF cookie's value in X-CSRF-Token (GO-HTTP-006), and bodies are
// bounded.
func (handler *Handler) packGuard(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method != http.MethodGet {
			cookie, err := r.Cookie(csrfCookieName)
			token := r.Header.Get("X-CSRF-Token")
			if err != nil || !validBrowserToken(token) || !validBrowserToken(cookie.Value) || subtle.ConstantTimeCompare([]byte(token), []byte(cookie.Value)) != 1 {
				handler.writeProblem(w, r, http.StatusForbidden, "CSRF_REJECTED", "The request could not be verified")
				return
			}
		}
		if handler.packBuilder == nil {
			handler.writeProblem(w, r, http.StatusServiceUnavailable, "PACK_BUILDER_UNAVAILABLE", "The pack builder is unavailable")
			return
		}
		r.Body = http.MaxBytesReader(w, r.Body, maxPackRequestBytes)
		next.ServeHTTP(w, r)
	})
}

// writePackBuilderError maps pack builder errors to problems; the specific
// conflicts come before the generic ones they wrap.
func (handler *Handler) writePackBuilderError(w http.ResponseWriter, r *http.Request, err error) {
	switch {
	case errors.Is(err, pack.ErrNameTaken):
		handler.writeProblem(w, r, http.StatusConflict, "PACK_NAME_TAKEN", "A pack with this name already exists")
	case errors.Is(err, pack.ErrStaleParent):
		handler.writeProblem(w, r, http.StatusConflict, "PACK_CHANGED", "The pack has a newer version; reload it and save again")
	case errors.Is(err, org.ErrInvalid):
		handler.writeProblem(w, r, http.StatusBadRequest, "INVALID_PACK", "The pack request is invalid")
	case errors.Is(err, org.ErrForbidden):
		handler.writeProblem(w, r, http.StatusForbidden, "FORBIDDEN", "Tenant administrator rights are required")
	case errors.Is(err, store.ErrNotFound):
		handler.writeProblem(w, r, http.StatusNotFound, "NOT_FOUND", "The requested pack is unavailable")
	default:
		handler.writeProblem(w, r, http.StatusInternalServerError, "PACK_FAILED", "The pack could not be saved or read", err)
	}
}
